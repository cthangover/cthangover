using System;
using System.Collections.Generic;
using System.Text.RegularExpressions;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;

namespace Cthangover.Authoring.CodeStyle
{
    /// <summary>Syntax-only accessibility and documentation rules; no assemblies or game resources are loaded.</summary>
    internal static class SyntaxContracts
    {
        internal static bool IsPublic(MemberDeclarationSyntax member)
        {
            var modifiers = ModifiersOf(member);
            if (modifiers.Any(SyntaxKind.PublicKeyword))
            {
                return true;
            }

            if (member.Parent is InterfaceDeclarationSyntax)
            {
                return !modifiers.Any(SyntaxKind.PrivateKeyword) && !modifiers.Any(SyntaxKind.InternalKeyword)
                    && !modifiers.Any(SyntaxKind.ProtectedKeyword);
            }

            return member is EnumMemberDeclarationSyntax && member.Parent is EnumDeclarationSyntax declaration
                && declaration.Modifiers.Any(SyntaxKind.PublicKeyword);
        }

        internal static bool HasDocumentation(SyntaxNode node)
        {
            foreach (var trivia in node.GetLeadingTrivia())
            {
                if (trivia.GetStructure()is not DocumentationCommentTriviaSyntax)
                {
                    continue;
                }

                var match = Regex.Match(trivia.ToFullString(), @"<summary\b[^>]*>([\s\S]*?)</summary>");
                if (match.Success && Regex.Replace(match.Groups[1].Value, @"<[^>]+>|///|\s", "").Length > 0)
                {
                    return true;
                }
            }

            return false;
        }

        internal static string TypeName(BaseTypeDeclarationSyntax declaration)
        {
            var names = new List<string>();
            foreach (var ancestor in declaration.Ancestors())
            {
                if (ancestor is BaseTypeDeclarationSyntax parent)
                {
                    names.Insert(0, parent.Identifier.ValueText);
                }
                else if (ancestor is BaseNamespaceDeclarationSyntax space)
                {
                    names.Insert(0, space.Name.ToString());
                }
            }

            names.Add(declaration.Identifier.ValueText);
            return string.Join(".", names);
        }

        internal static string SimpleTypeName(TypeSyntax type)
        {
            return type switch
            {
                IdentifierNameSyntax identifier => identifier.Identifier.ValueText,
                GenericNameSyntax generic => generic.Identifier.ValueText,
                QualifiedNameSyntax qualified => SimpleTypeName(qualified.Right),
                AliasQualifiedNameSyntax alias => SimpleTypeName(alias.Name),
                _ => type.ToString(),
            };
        }

        internal static bool IsStatement(StatementSyntax statement)
        {
            return statement is ExpressionStatementSyntax
                or LocalDeclarationStatementSyntax
                or ReturnStatementSyntax
                or ThrowStatementSyntax
                or BreakStatementSyntax
                or ContinueStatementSyntax
                or GotoStatementSyntax
                or YieldStatementSyntax
                or EmptyStatementSyntax;
        }

        private static SyntaxTokenList ModifiersOf(MemberDeclarationSyntax member)
        {
            return member switch
            {
                BaseTypeDeclarationSyntax type => type.Modifiers,
                DelegateDeclarationSyntax type => type.Modifiers,
                BaseMethodDeclarationSyntax method => method.Modifiers,
                BasePropertyDeclarationSyntax property => property.Modifiers,
                BaseFieldDeclarationSyntax field => field.Modifiers,
                _ => default,
            };
        }
    }
}
