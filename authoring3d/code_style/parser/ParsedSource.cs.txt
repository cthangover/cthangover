using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;

namespace Cthangover.Authoring.CodeStyle
{
    /// <summary>Owns a single syntax tree for the duration of a batch; no compilation or Godot runtime is required.</summary>
    internal sealed class ParsedSource
    {
        internal SourceInput Input { get; }
        internal SyntaxTree Tree { get; }
        internal CompilationUnitSyntax Root { get; }

        internal ParsedSource(SourceInput input)
        {
            Input = input;
            Tree = CSharpSyntaxTree.ParseText(input.Text, new CSharpParseOptions(documentationMode: DocumentationMode.Parse));
            Root = Tree.GetCompilationUnitRoot();
        }
    }
}
