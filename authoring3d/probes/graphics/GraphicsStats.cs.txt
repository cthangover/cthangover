namespace Cthangover.Probes.Graphics
{
    using System;using System.Collections.Generic;using System.Text.Json.Nodes;
    /// <summary>Nearest-rank quantiles; the raw per-frame samples accompany every distribution.</summary>
    public sealed class GraphicsStats
    {
        private readonly List<double> data=new();
        public void Add(double value){data.Add(value);}
        public JsonObject Json(){if(data.Count==0)return new(){["n"]=0};data.Sort();double total=0;foreach(var v in data)total+=v;return new(){["n"]=data.Count,["mean"]=total/data.Count,["p50"]=data[(data.Count-1)/2],["p95"]=data[Math.Min(data.Count-1,(int)Math.Ceiling(data.Count*.95)-1)],["p99"]=data[Math.Min(data.Count-1,(int)Math.Ceiling(data.Count*.99)-1)],["max"]=data[^1]};}
    }
}
