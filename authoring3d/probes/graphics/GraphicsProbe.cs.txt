        public override void _Process(double delta)
        {
            var now=lifetime.Elapsed.TotalMilliseconds;if(lastFrame>0&&entrance!=null)entrance.Add(now-lastFrame);lastFrame=now;if(!active)return;
            try{elapsed+=delta;var watch=Stopwatch.StartNew();foreach(var actor in actors){actor.Advance(delta);if(actor.Animation.Clock.Position>=actor.Animation.Clips.Clips[actor.Animation.CurrentState].Animation.Length)actor.Animation.PlayState(actor.Animation.CurrentState);}animation.Add(watch.Elapsed.TotalMilliseconds);var t=(float)elapsed;camera.Position=new Vector3(22+5*Mathf.Sin(t*.5f),20,36+3*Mathf.Cos(t*.5f));camera.LookAt(new Vector3(-4,3,-6));terrain.Morph(camera.Position);}
            catch(Exception e){updateError=e;active=false;}
        }
        public override void _PhysicsProcess(double delta)
        {
            if(!active||++physicsFrame%simulationStride!=0)return;var watch=Stopwatch.StartNew();for(var i=0;i<bodies.Count;i++){var body=bodies[i];var direction=new Vector3(Mathf.Sin((float)elapsed*.5f+i),0,Mathf.Cos((float)elapsed*.5f+i));var v=direction*1.2f;v.Y=body.IsOnFloor()?-.1f:body.Velocity.Y-9.8f*(float)delta;body.Velocity=v;body.MoveAndSlide();}move.Add(watch.Elapsed.TotalMilliseconds);
        }

        {
            var wall=new GraphicsStats();var main=new GraphicsStats();var render=new GraphicsStats();var gpu=new GraphicsStats();var physics=new GraphicsStats();var samples=new JsonArray();var timer=Stopwatch.StartNew();var previous=timer.Elapsed.TotalMilliseconds;
            do{await Frames(1);if(updateError!=null)throw updateError;var now=timer.Elapsed.TotalMilliseconds;var dt=now-previous;previous=now;var m=Performance.GetMonitor(Performance.Monitor.TimeProcess)*1000;var r=RenderingServer.ViewportGetMeasuredRenderTimeCpu(viewport.GetViewportRid())+RenderingServer.ViewportGetMeasuredRenderTimeCpu(GetViewport().GetViewportRid())+RenderingServer.GetFrameSetupTimeCpu();var g=RenderingServer.ViewportGetMeasuredRenderTimeGpu(viewport.GetViewportRid())+RenderingServer.ViewportGetMeasuredRenderTimeGpu(GetViewport().GetViewportRid());var p=Performance.GetMonitor(Performance.Monitor.TimePhysicsProcess)*1000;wall.Add(dt);main.Add(m);render.Add(r);gpu.Add(g);physics.Add(p);samples.Add(new JsonArray(dt,m,r,g,p));}while((timer.Elapsed.TotalSeconds<seconds||samples.Count<minimum)&&samples.Count<4000);

        private void Apply(string key,bool on)
        {
            switch(key)
            {
                case "scale_bilinear":viewport.Scaling3DScale=on?.67f:1;break;
                case "fsr1":viewport.Scaling3DMode=on?Viewport.Scaling3DModeEnum.Fsr:Viewport.Scaling3DModeEnum.Bilinear;viewport.Scaling3DScale=on?.67f:1;break;
                case "fsr2":viewport.Scaling3DMode=on?Viewport.Scaling3DModeEnum.Fsr2:Viewport.Scaling3DModeEnum.Bilinear;viewport.Scaling3DScale=on?.67f:1;break;
                case "msaa":viewport.Msaa3D=on?Viewport.Msaa.Msaa4X:Viewport.Msaa.Disabled;break;
                case "fxaa":viewport.ScreenSpaceAA=on?Viewport.ScreenSpaceAAEnum.Fxaa:Viewport.ScreenSpaceAAEnum.Disabled;break;
                case "taa":viewport.UseTaa=on;break;
                case "shadows":sun.ShadowEnabled=!on;break;
                case "ssao":environment.SsaoEnabled=on;break;
                case "ssil":environment.SsilEnabled=on;break;
                case "sdfgi":environment.SdfgiMinCellSize=4;environment.SdfgiCascades=1;environment.SdfgiEnabled=on;break;
                case "volumetric_fog":environment.VolumetricFogEnabled=on;environment.VolumetricFogDensity=.01f;break;
                case "voxelgi":if(on){voxel=new VoxelGI{Size=new Vector3(48,24,48),Subdiv=VoxelGI.SubdivEnum.Subdiv64};viewport.AddChild(voxel);voxel.Bake(viewport,false);voxelData=voxel.Data;}else if(voxel!=null){voxel.Visible=false;voxel.Data=null;voxel.Free();voxel=null;voxelData?.Dispose();voxelData=null;}break;
                case "mesh_lod":viewport.MeshLodThreshold=on?8:1;break;
                case "distance":camera.Far=on?60:500;break;
                case "grass":foreach(var mesh in grass)mesh.VisibleInstanceCount=on?mesh.InstanceCount/2:-1;break;
                case "animation_hz":foreach(var actor in actors)actor.SetDistance(on?30:0);break;
                case "simulation_lod":simulationStride=on?3:1;break;
                case "vsync":DisplayServer.WindowSetVsyncMode(on?DisplayServer.VSyncMode.Enabled:DisplayServer.VSyncMode.Disabled);break;
                case "fps_cap":Engine.MaxFps=on?30:0;break;
            }
        }
