using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using System.Text.Json;
using System.Text.RegularExpressions;
/// <summary>One portable component checks editor UI sinks and locale references through the C# syntax tree, including mutation probes for the checker itself. No Godot or whole-game tests are run.</summary>
public static class EditorLocaleCheck
{
    private static readonly string[] Modules={"world3d_editor","anim3d_editor","characters3d_editor","scene3d_editor","characters3d_base","anim3d_base"};
    private static readonly HashSet<string> Properties=new(){"Text","TooltipText","PlaceholderText","Title","Prefix","Suffix"};
    private static readonly HashSet<string> Calls=new(){"SetText","SetTooltipText","SetColumnTitle","SetTabTitle","AddItem","SetItemText","DrawString","Describe","SetStatus","SetHint","ShowError"};
    private static readonly HashSet<string> Translation=new(){"Tr","Get","Translate","Caption"};
    public static int Main(string[] args)
    {
        var root=Path.GetFullPath(args.Length>0?args[0]:".");var errors=new List<string>();var all=new Dictionary<string,string>();var counts=new Dictionary<string,int>();
        foreach(var mod in Modules)
        {
            var ru=Read(Path.Combine(root,"mods_private",mod,"locale","ru.properties"),errors);var en=Read(Path.Combine(root,"mods_private",mod,"locale","en.properties"),errors);counts[mod]=ru.Count;
            foreach(var key in ru.Keys.Union(en.Keys))
            {
                if(!HasBoth(key,ru,en)){errors.Add(mod+": missing ru/en key "+key);continue;}
                if(string.IsNullOrWhiteSpace(ru[key])||string.IsNullOrWhiteSpace(en[key]))errors.Add(mod+": empty key "+key);
                if(Regex.IsMatch(en[key],"[А-Яа-яЁё]"))errors.Add(mod+": Cyrillic in en "+key);
                if(Placeholders(ru[key])!=Placeholders(en[key]))errors.Add(mod+": placeholder mismatch "+key);
                all[key]=ru[key];
            }
        }
        foreach(var mod in Modules)
        {
            foreach(var path in Directory.EnumerateFiles(Path.Combine(root,"mods_private",mod,"src"),"*.cs"))
            {
                var source=File.ReadAllText(path);var tree=CSharpSyntaxTree.ParseText(source);foreach(var d in tree.GetDiagnostics())if(d.Severity==DiagnosticSeverity.Error)errors.Add(path+": "+d);
                foreach(var literal in tree.GetRoot().DescendantNodes().OfType<LiteralExpressionSyntax>())
                {
                    if(!literal.IsKind(SyntaxKind.StringLiteralExpression))continue;var text=literal.Token.ValueText;
                    if(UiKey(text)&&!DynamicPrefix(literal,text)&&!all.ContainsKey(text))errors.Add(path+":"+(literal.GetLocation().GetLineSpan().StartLinePosition.Line+1)+": undeclared locale key "+text);
                    if(mod.EndsWith("_editor",StringComparison.Ordinal)&&Hardcoded(literal))errors.Add(path+":"+(literal.GetLocation().GetLineSpan().StartLinePosition.Line+1)+": visible literal "+text);
                }
            }
        }
        if(CheckSnippet("new Label { Text = \"Untranslated caption\" };")!=1||CheckSnippet("new Label { Text = Tr(\"anim3d/help\") };")!=0||CheckSnippet("tree.SetColumnTitle(0, \"Причина\");")!=1||CheckSnippet("label.Text = data[\"Role\"].ToString();")!=0)errors.Add("checker mutation probes failed");
        if(HasBoth("probe",new Dictionary<string,string>{{"probe","ru"}},new Dictionary<string,string>())||!HasBoth("probe",new Dictionary<string,string>{{"probe","ru"}},new Dictionary<string,string>{{"probe","en"}}))errors.Add("missing-language mutation probes failed");
        Console.WriteLine(JsonSerializer.Serialize(new Dictionary<string,object>{{"modules",counts},{"errors",errors},{"mutationProbes",6}}));
        return errors.Count==0?0:1;
    }
    private static bool HasBoth(string key,Dictionary<string,string> ru,Dictionary<string,string> en)=>ru.ContainsKey(key)&&en.ContainsKey(key);
    private static Dictionary<string,string> Read(string path,List<string> errors)
    {
        var result=new Dictionary<string,string>();foreach(var line in File.ReadAllLines(path))
        {var text=line.Trim();if(text.Length==0||text.StartsWith('#'))continue;var n=text.IndexOf('=');if(n<0){errors.Add(path+": invalid properties line");continue;}var key=text[..n].Trim();if(!result.TryAdd(key,text[(n+1)..].Trim()))errors.Add(path+": duplicate key "+key);}return result;
    }
    private static string Placeholders(string value){var items=new SortedSet<string>();foreach(Match m in Regex.Matches(value,@"\{\d+\}"))items.Add(m.Value);return string.Join(",",items);}
    private static bool UiKey(string value)=>value.StartsWith("anim3d/",StringComparison.Ordinal)||value.StartsWith("characters3d/",StringComparison.Ordinal)||value.StartsWith("tools/world3d_editor/",StringComparison.Ordinal)||value.StartsWith("tools/scene3d_editor/",StringComparison.Ordinal);
    private static bool DynamicPrefix(LiteralExpressionSyntax node,string value)=>value.EndsWith('/')||value.EndsWith('_')||node.Parent is BinaryExpressionSyntax||value=="characters3d/instance";
    private static string Name(ExpressionSyntax expression)=>expression is MemberAccessExpressionSyntax member?member.Name.Identifier.Text:expression is IdentifierNameSyntax id?id.Identifier.Text:"";
    private static bool Hardcoded(LiteralExpressionSyntax literal)
    {
        var value=literal.Token.ValueText;if(!Regex.IsMatch(value,"[A-Za-zА-Яа-яЁё]"))return false;
        // Draft IDs and mathematical axis tokens are data, not translated captions.
        if(value is "X" or "Y" or "Z" or "-X" or "-Y" or "-Z")return false;
        if(value=="clip"||value=="new_")foreach(var parent in literal.Ancestors())if(parent is AssignmentExpressionSyntax draft){if(draft.Left is MemberAccessExpressionSyntax field&&field.Expression is IdentifierNameSyntax variable&&(variable.Identifier.Text=="name"||variable.Identifier.Text=="id"))return false;break;}
        foreach(var node in literal.Ancestors())
        {
            if(node is ElementAccessExpressionSyntax or ElementBindingExpressionSyntax)return false;
            if(node is InvocationExpressionSyntax invoke)
            {var name=Name(invoke.Expression);if(Translation.Contains(name))return false;if(Calls.Contains(name))return true;return false;}
            if(node is AssignmentExpressionSyntax assignment)return Properties.Contains(Name(assignment.Left));
            if(node is BinaryExpressionSyntax binary && (binary.IsKind(SyntaxKind.EqualsExpression)||binary.IsKind(SyntaxKind.NotEqualsExpression)))return false;
            if(node is StatementSyntax)return false;
        }
        return false;
    }
    private static int CheckSnippet(string source)
    {var count=0;foreach(var node in CSharpSyntaxTree.ParseText(source).GetRoot().DescendantNodes().OfType<LiteralExpressionSyntax>())if(node.IsKind(SyntaxKind.StringLiteralExpression)&&Hardcoded(node))count++;return count;}
}
